cmake_minimum_required(VERSION 3.16)
project(etrace_diag LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(cmake/libbpf.cmake)
include(cmake/bpftool.cmake)

# ---------------------------------------------------------------------------
# BPF artifact generation: vmlinux.h -> etrace.bpf.o -> etrace.skel.h
# ---------------------------------------------------------------------------
set(ETD_BPF_DIR ${CMAKE_CURRENT_SOURCE_DIR}/bpf)
set(ETD_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/bpf)
file(MAKE_DIRECTORY ${ETD_GEN_DIR})

set(VMLINUX_H ${ETD_GEN_DIR}/vmlinux.h)
add_custom_command(
  OUTPUT ${VMLINUX_H}
  COMMAND ${ETD_BPFTOOL} btf dump file /sys/kernel/btf/vmlinux format c > ${VMLINUX_H}
  DEPENDS ${ETD_BPFTOOL}
  COMMENT "Generating vmlinux.h from /sys/kernel/btf/vmlinux"
  VERBATIM)

set(BPF_OBJ ${ETD_GEN_DIR}/etrace.bpf.o)
set(BPF_SKEL ${ETD_GEN_DIR}/etrace.skel.h)

# Compose -I flags for the BPF compile.
set(ETD_BPF_INCLUDES "")
foreach(d ${ETD_LIBBPF_INCLUDE_DIRS})
  list(APPEND ETD_BPF_INCLUDES -I${d})
  list(APPEND ETD_BPF_INCLUDES -I${d}/uapi)
endforeach()
list(APPEND ETD_BPF_INCLUDES -I${ETD_BPF_DIR})
list(APPEND ETD_BPF_INCLUDES -I${ETD_GEN_DIR})

# Kernel version + feature detection (shared by BPF compile and C++ host).
# vmlinux.h is generated from the RUNNING kernel's BTF at build time, so the
# running kernel version decides which feature implementations to compile in.
execute_process(COMMAND uname -r OUTPUT_VARIABLE ETD_KERNEL_REL
                OUTPUT_STRIP_TRAILING_WHITESPACE)
set(ETD_KERNEL_MAJOR 6)
set(ETD_KERNEL_MINOR 6)
if(ETD_KERNEL_REL MATCHES "^([0-9]+)\\.([0-9]+)")
  set(ETD_KERNEL_MAJOR ${CMAKE_MATCH_1})
  set(ETD_KERNEL_MINOR ${CMAKE_MATCH_2})
endif()

set(ETD_BPF_EXTRA_DEFINES
    -DETD_KERNEL_MAJOR=${ETD_KERNEL_MAJOR}
    -DETD_KERNEL_MINOR=${ETD_KERNEL_MINOR})
set(ETD_HOST_DEFINES
    ETD_KERNEL_MAJOR=${ETD_KERNEL_MAJOR}
    ETD_KERNEL_MINOR=${ETD_KERNEL_MINOR})
# struct bpf_timer in vmlinux BTF, and lock:contention_begin/end tracepoints,
# are both only available on >= 6.0.
if(ETD_KERNEL_MAJOR GREATER_EQUAL 6)
  list(APPEND ETD_BPF_EXTRA_DEFINES -DETD_HAVE_BPF_TIMER_UAPI -DETD_HAVE_LOCK_CONTENTION_TP)
  list(APPEND ETD_HOST_DEFINES ETD_HAVE_LOCK_CONTENTION_TP)
endif()

# bpf_map_lookup_percpu_elem (cross-CPU percpu read) exists only on >= 6.4.
# When absent, the flight recorder falls back to a per-CPU perf sampler and the
# host merges per-CPU records (see bpf/etrace.bpf.c + ebpf_manager).
set(ETD_HAVE_PERCPU_LOOKUP_ELEM FALSE)
if(ETD_KERNEL_MAJOR GREATER 6 OR (ETD_KERNEL_MAJOR EQUAL 6 AND ETD_KERNEL_MINOR GREATER_EQUAL 4))
  set(ETD_HAVE_PERCPU_LOOKUP_ELEM TRUE)
  list(APPEND ETD_BPF_EXTRA_DEFINES -DETD_HAVE_PERCPU_LOOKUP_ELEM)
  list(APPEND ETD_HOST_DEFINES ETD_HAVE_PERCPU_LOOKUP_ELEM)
endif()

if(ETD_HAVE_PERCPU_LOOKUP_ELEM)
  message(STATUS "etrace-diag: percpu-lookup-elem available -> summed flight recorder")
else()
  message(STATUS "etrace-diag: percpu-lookup-elem unavailable -> per-CPU sampler fallback")
endif()

add_custom_command(
  OUTPUT ${BPF_OBJ}
  COMMAND ${ETD_CLANG} -target bpf -O2 -g ${ETD_BPF_ARCH_DEFINE} -mcpu=v3
          -D__KERNEL__ ${ETD_BPF_EXTRA_DEFINES}
          ${ETD_BPF_INCLUDES}
          -c ${ETD_BPF_DIR}/etrace.bpf.c -o ${BPF_OBJ}
  DEPENDS ${ETD_BPF_DIR}/etrace.bpf.c ${VMLINUX_H} ${ETD_LIBBPF_LIBRARIES}
  COMMENT "Compiling etrace.bpf.c -> etrace.bpf.o"
  VERBATIM)

add_custom_command(
  OUTPUT ${BPF_SKEL}
  COMMAND ${ETD_BPFTOOL} gen skeleton ${BPF_OBJ} > ${BPF_SKEL}
  DEPENDS ${BPF_OBJ} ${ETD_BPFTOOL}
  COMMENT "Generating etrace.skel.h via bpftool gen skeleton"
  VERBATIM)

add_custom_target(bpf_artifacts ALL DEPENDS ${BPF_SKEL})

# ---------------------------------------------------------------------------
# User-space sources
# ---------------------------------------------------------------------------
file(GLOB_RECURSE ETD_SOURCES CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)

add_executable(etrace-diag ${ETD_SOURCES})

target_include_directories(etrace-diag SYSTEM PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/third_party
  ${ETD_LIBBPF_INCLUDE_DIRS})

target_include_directories(etrace-diag PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/include
  ${CMAKE_CURRENT_SOURCE_DIR}/src
  ${ETD_GEN_DIR})

target_compile_options(etrace-diag PRIVATE -O2 -Wall -Werror)
target_compile_definitions(etrace-diag PRIVATE ${ETD_HOST_DEFINES})

# The skeleton header is generated by bpftool; ensure it exists before compiling.
if(TARGET libbpf_build)
  add_dependencies(etrace-diag libbpf_build)
endif()
add_dependencies(etrace-diag bpf_artifacts)

find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)

target_link_directories(etrace-diag PRIVATE ${ETD_LIBBPF_LIBRARY_DIRS})

target_link_libraries(etrace-diag PRIVATE
  ${ETD_LIBBPF_LIBRARIES}
  elf
  z
  ${CMAKE_DL_LIBS}
  Threads::Threads
  SQLite::SQLite3)

  ${CMAKE_CURRENT_SOURCE_DIR}/src/config.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/src/logging.cpp)

  ${CMAKE_CURRENT_SOURCE_DIR}/include
  ${CMAKE_CURRENT_SOURCE_DIR}/src
  ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
target_compile_options(config_probe PRIVATE -O2 -Wall -Werror)
add_test(NAME config_probe COMMAND config_probe
